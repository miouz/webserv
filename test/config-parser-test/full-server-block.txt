server {
  listen 8080;
  server_name mysite.com;
  root /var/www;
  index index.html index.htm;
  client_max_body_size 2048;
  error_page 404 /404.html;
  error_page 500 /500.html;
  location / {
    methods GET POST;
    autoindex off;
  }
  location /upload {
    methods POST;
    upload_store /tmp;
    client_max_body_size 8192;
  }
}
